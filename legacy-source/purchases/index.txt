'use strict';

async function loadItems() {
  items = await DB.getAll('purchases');
}

async function loadCategories() {
  // categories stored as purchases-scoped entries
  // using DB tags collection isn't ideal — use our own collection via metadata
  // We'll store categories in 'categories' collection with module:'purchases'
  const all = await DB.getAll('categories').catch(() => []);
  categories = all.filter(c => !c.module || c.module === 'purchases');
  if (categories.length === 0) {
    // seed defaults
    const defaults = [
      { name: 'Техника', emoji: '💻' }, { name: 'Одежда', emoji: '👗' },
      { name: 'Кухня', emoji: '🍳' },   { name: 'Косплей', emoji: '🎭' },
      { name: 'Красота', emoji: '💅' },  { name: 'Дом', emoji: '🏠' },
      { name: 'Спорт', emoji: '🏋️' },   { name: 'Книги', emoji: '📚' },
    ];
    for (const d of defaults) {
      const rec = await DB.create('categories', { ...d, module: 'purchases' });
      categories.push(rec);
    }
  }
}

async function addCategory(name, emoji) {
  const rec = await DB.create('categories', { name, emoji, module: 'purchases' });
  categories.push(rec);
  return rec;
}

async function deleteCategory(id) {
  await DB.delete('categories', id);
  categories = categories.filter(c => c.id !== id);
}

function fmtPrice(n) {
  if (n == null || n === '') return '—';
  return Number(n).toLocaleString('ru-RU') + ' ₽';
}

function getVisible() {
  let list = [...items];
  if (filterStatus !== 'all') list = list.filter(i => i.status === filterStatus);
  if (filterSource !== 'all') list = list.filter(i => (i.sourceType || 'manual') === filterSource);
  if (searchQ) {
    const q = searchQ.toLowerCase();
    list = list.filter(i =>
      (i.name||'').toLowerCase().includes(q) ||
      (i.category||'').toLowerCase().includes(q) ||
      (i.tags||[]).some(t => t.toLowerCase().includes(q)) ||
      (i.store||'').toLowerCase().includes(q)
    );
  }
  switch (sortMode) {
    case 'price_asc':  list.sort((a,b) => (a.price||0)-(b.price||0)); break;
    case 'price_desc': list.sort((a,b) => (b.price||0)-(a.price||0)); break;
    case 'date':       list.sort((a,b) => new Date(b.createdAt)-new Date(a.createdAt)); break;
    default:           list.sort((a,b) => (b.priority||0)-(a.priority||0)); break;
  }
  return list;
}

function render() {
  loadingEl.style.display = 'none';
  const list = getVisible();

  // stats from all items (no filter)
  const notDone = items.filter(i => i.status !== 'in_stock');
  $('stTotal').textContent = items.length;
  $('stSum').textContent   = notDone.length ? fmtPrice(notDone.reduce((s,i) => s + (i.price||0), 0)) : '—';
  $('stDone').textContent  = items.filter(i => i.status === 'in_stock').length;
  $('stLeft').textContent  = notDone.length;

  if (list.length === 0) {
    mainList.style.display = 'none';
    emptyEl.style.display  = 'flex';
    return;
  }
  emptyEl.style.display  = 'none';
  mainList.style.display = 'block';

  if (viewMode === 'gallery') renderGallery(list);
  else renderList(list);
}

function srcLabel(src) {
  const map = { manual: '', task: 'Задача', cosplay: 'Косплей', wishlist: 'Вишлист', meal_plan: 'Питание' };
  return map[src] || '';
}

async function thumbHTML(item) {
  if (item.imageIds && item.imageIds.length) {
    const img = await DB.getImage(item.imageIds[0]);
    if (img) return `<img src="${img.data}" loading="lazy"/>`;
  }
  // emoji fallback by category
  const catEmojis = { 'Техника':'💻','Одежда':'👗','Кухня':'🍳','Косплей':'🎭','Красота':'💅','Дом':'🏠','Спорт':'🏋️','Книги':'📚' };
  return catEmojis[item.category] || '🛍️';
}

async function renderGallery(list) {
  mainList.innerHTML = `<div class="gal-grid" style="--cols:${gridCols}" id="galGrid"></div>`;
  const grid = $('galGrid');
  for (let idx = 0; idx < list.length; idx++) {
    const item = list[idx];
    const el = document.createElement('div');
    el.className = 'gal-card';
    el.dataset.prio = item.priority ?? 1;
    el.dataset.id   = item.id;
    el.style.animationDelay = `${idx * 0.02}s`;
    const src = srcLabel(item.sourceType);
    const th  = await thumbHTML(item);
    el.innerHTML = `
      <div class="gal-img">
        ${th}
        <div class="gal-prio-dot"></div>
        ${src ? `<div class="gal-src-badge">${src}</div>` : ''}
      </div>
      <div class="gal-body">
        <div class="gal-name">${esc(item.name)}</div>
        ${item.price ? `<div class="gal-price">${fmtPrice(item.price)}</div>` : ''}
      </div>`;
    el.addEventListener('click', () => openDetail(item.id));
    grid.appendChild(el);
  }
}

async function renderList(list) {
  const groups = {};
  list.forEach(item => {
    const cat = item.category || '—';
    if (!groups[cat]) groups[cat] = [];
    groups[cat].push(item);
  });

  mainList.innerHTML = '';
  for (const [cat, grpItems] of Object.entries(groups)) {
    const catSum   = grpItems.filter(i => i.status !== 'in_stock').reduce((s,i) => s+(i.price||0), 0);
    const catObj   = categories.find(c => c.name === cat);
    const catEmoji = catObj ? catObj.emoji : '📦';

    const group = document.createElement('div');
    group.className = 'cat-group';

    let cardsHtml = '';
    for (let idx = 0; idx < grpItems.length; idx++) {
      const item = grpItems[idx];
      const p = item.priority ?? 1;
      const pLabel = p===2 ? ['p2','🔴'] : p===1 ? ['p1','📅'] : ['p0','○'];
      const dotCls = item.status==='ordered' ? 'ordered' : item.status==='in_stock' ? 'in_stock' : '';
      const th = await thumbHTML(item);
      const src = srcLabel(item.sourceType);
      cardsHtml += `<div class="p-card ${item.status==='in_stock'?'done':''}" data-prio="${p}" data-id="${item.id}" style="animation-delay:${idx*0.025}s">
        <div class="p-thumb">${th}</div>
        <div class="p-body">
          <div class="p-name">${esc(item.name)}</div>
          <div class="p-meta-row">
            ${item.price ? `<span class="p-price">${fmtPrice(item.price)}</span>` : ''}
            <span class="p-status-dot ${dotCls}"></span>
            ${src ? `<span class="p-source">${src}</span>` : ''}
            ${item.store ? `<span class="p-source">@ ${esc(item.store)}</span>` : ''}
          </div>
          ${(item.tags||[]).length ? `<div style="display:flex;gap:3px;flex-wrap:wrap;margin-top:4px">${(item.tags||[]).map(t=>`<span style="padding:1px 7px;border-radius:100px;font-size:10px;background:var(--bg3);color:var(--text3)">${esc(t)}</span>`).join('')}</div>` : ''}
        </div>
        <div class="p-right">
          <span class="p-prio-dot ${pLabel[0]}">${pLabel[1]}</span>
        </div>
      </div>`;
    }

    group.innerHTML = `
      <div class="cat-header">
        <div class="cat-name">${cat==='—' ? '📦 Без категории' : `${catEmoji} ${esc(cat)}`}</div>
        <div class="cat-meta">${grpItems.length} шт · <span class="cat-sum">${fmtPrice(catSum)}</span></div>
      </div>
      ${cardsHtml}`;

    group.querySelectorAll('[data-id]').forEach(el => {
      el.addEventListener('click', () => openDetail(el.dataset.id));
    });
    mainList.appendChild(group);
  }
}

async function openAnalytics() {
  const body = $('analyticsBody');
  body.innerHTML = '<div class="state-center"><div class="spinner"></div></div>';
  $('analyticsPanel').classList.add('open');

  const all = await DB.getAll('purchases');
  const total   = all.length;
  const bought  = all.filter(i => i.status === 'in_stock');
  const pending = all.filter(i => i.status !== 'in_stock');
  const totalSpend   = bought.reduce((s,i) => s+(i.price||0), 0);
  const plannedSpend = pending.reduce((s,i) => s+(i.price||0), 0);

  // by category
  const byCat = {};
  all.forEach(i => { const c = i.category||'Без категории'; byCat[c] = (byCat[c]||0) + (i.price||0); });
  const catEntries = Object.entries(byCat).sort((a,b)=>b[1]-a[1]);
  const maxCat = catEntries[0]?.[1] || 1;

  // by source
  const bySrc = {};
  all.forEach(i => { const s = i.sourceType||'manual'; bySrc[s] = (bySrc[s]||0) + 1; });

  // conversion wish→bought
  const wishTotal = all.filter(i => i.sourceType !== 'cosplay').length;
  const convRate  = total ? Math.round(bought.length / total * 100) : 0;

  body.innerHTML = `
    <div class="an-section">
      <div class="an-section-title">Общая статистика</div>
      <div class="an-stat-grid">
        <div class="an-stat accent">
          <div class="an-stat-val">${total}</div>
          <div class="an-stat-lbl">всего позиций</div>
        </div>
        <div class="an-stat green">
          <div class="an-stat-val">${fmtPrice(totalSpend)}</div>
          <div class="an-stat-lbl">потрачено</div>
        </div>
        <div class="an-stat yellow">
          <div class="an-stat-val">${fmtPrice(plannedSpend)}</div>
          <div class="an-stat-lbl">запланировано</div>
        </div>
        <div class="an-stat red">
          <div class="an-stat-val">${convRate}%</div>
          <div class="an-stat-lbl">куплено от списка</div>
        </div>
      </div>
    </div>

    <div class="an-section">
      <div class="an-section-title">По категориям</div>
      ${catEntries.slice(0, 8).map(([name, sum]) => `
        <div class="an-bar-row">
          <div class="an-bar-label">
            <span style="font-size:12px">${esc(name)}</span>
            <span style="font-size:11px;color:var(--text3);font-family:var(--m)">${fmtPrice(sum)}</span>
          </div>
          <div class="an-bar-track">
            <div class="an-bar-fill" style="width:${Math.round(sum/maxCat*100)}%"></div>
          </div>
        </div>`).join('')}
    </div>

    <div class="an-section" style="padding-bottom:20px">
      <div class="an-section-title">По источникам</div>
      <div style="display:flex;flex-direction:column;gap:8px">
        ${Object.entries(bySrc).map(([src, cnt]) => {
          const srcNames = { manual:'✋ Ручные', task:'📋 Из задач', cosplay:'🎭 Косплей', wishlist:'💛 Вишлист', meal_plan:'🍽 Питание' };
          return `<div style="display:flex;justify-content:space-between;padding:9px 12px;background:var(--bg3);border-radius:8px;font-size:13px">
            <span>${srcNames[src]||src}</span>
            <span style="font-family:var(--m);color:var(--accent2)">${cnt}</span>
          </div>`;
        }).join('')}
      </div>
    </div>
  `;
}

function resetForm() {
  editId = null; imgData = null; imgId = null;
  curTags = []; curPriority = 1; curStatus = 'wish'; curSource = 'manual';

  $('tagIn').value = '';

async function openEdit(id) {
  const item = items.find(i => i.id === id);
  if (!item) return;
  resetForm();
  editId     = id;
  curTags    = [...(item.tags||[])];
  curPriority = item.priority ?? 1;
  curStatus   = item.status   || 'wish';
  curSource   = item.sourceType || 'manual';
  imgId       = (item.imageIds && item.imageIds[0]) || null;

  $('fName').value  = item.name  || '';
  $('fPrice').value = item.price || '';
  $('fCat').value   = item.category || '';
  $('fUrl').value   = item.url   || '';
  $('fNotes').value = item.notes || '';
  $('fStore').value = item.store || '';

  if (imgId) {
    const img = await DB.getImage(imgId);
    if (img) {
      $('imgPreview').src = img.data;
      $('imgPreview').style.display = 'block';
      $('imgUpLabel').style.display = 'none';
    }
  }

  renderTagChips(); syncPrioBtns(); syncStatusBtns(); syncSourceBtns();
  $('formTitle').textContent = 'Редактировать';
  $('btnDel').style.display = 'block';
  openSheet('form');
}

$('btnSave').addEventListener('click', async () => {
  const name = $('fName').value.trim();
  if (!name) { UI.toast('Введи название', 'err'); return; }
  const btn = $('btnSave');
  btn.disabled = true; btn.textContent = 'Сохранение...';
  try {
    // handle new image
    let imageIds = editId ? (items.find(i=>i.id===editId)?.imageIds||[]) : [];
    if (imgData) {
      const newId = await DB.saveImage(imgData, { name: 'purchase-photo' });
      imageIds = [newId]; // replace with new single image
    }

    const catVal = $('fCat').value.trim();
    if (catVal && !categories.find(c => c.name === catVal)) {
      await addCategory(catVal, '📦');
    }

    const payload = {
      name,
      price:      parseFloat($('fPrice').value) || null,
      category:   catVal || null,
      url:        $('fUrl').value.trim() || null,
      notes:      $('fNotes').value.trim() || null,
      store:      $('fStore').value.trim() || null,
      tags:       curTags,
      priority:   curPriority,
      status:     curStatus,
      sourceType: curSource,
      imageIds,
    };

    if (editId) {
      await DB.update('purchases', editId, payload);
    } else {
      await DB.create('purchases', payload);
    }
    await loadItems(); render(); closeSheets();
    UI.toast(editId ? 'Обновлено ✓' : 'Добавлено ✓', 'ok');
  } catch(e) {
    UI.toast('Ошибка: ' + e.message, 'err');
  } finally {
    btn.disabled = false; btn.textContent = 'Сохранить';
  }
});

$('btnDel').addEventListener('click', async () => {
  if (!editId || !confirm('Удалить эту покупку?')) return;
  await DB.delete('purchases', editId);
  items = items.filter(i => i.id !== editId);
  render(); closeSheets(); UI.toast('Удалено', 'ok');
});

const tagIn = $('tagIn');
tagIn.addEventListener('keydown', e => {
  if (e.key==='Enter'||e.key===',') { e.preventDefault(); addTag(tagIn.value.trim()); }
  if (e.key==='Backspace'&&!tagIn.value&&curTags.length) { curTags.pop(); renderTagChips(); }
});
tagIn.addEventListener('blur', () => { if(tagIn.value.trim()) addTag(tagIn.value.trim()); });
$('tagsWrap').addEventListener('click', () => tagIn.focus());

function addTag(v) {
  if (!v || curTags.includes(v)) { tagIn.value=''; return; }
  curTags.push(v); tagIn.value=''; renderTagChips();
}
function renderTagChips() {
  $('tagsWrap').querySelectorAll('.tc').forEach(el => el.remove());
  curTags.forEach((t, i) => {
    const chip = document.createElement('span');
    chip.className = 'tc';
    chip.innerHTML = `${esc(t)}<button data-i="${i}">×</button>`;
    chip.querySelector('button').addEventListener('click', () => { curTags.splice(i, 1); renderTagChips(); });
    $('tagsWrap').insertBefore(chip, tagIn);
  });
}

function renderCatDropdown(q) {
  const dd = $('catDropdown');
  const filtered = categories.filter(c => !q || c.name.toLowerCase().includes(q.toLowerCase()));
  const showNew  = q && !categories.find(c => c.name.toLowerCase()===q.toLowerCase());

  dd.innerHTML = filtered.map(c =>
    `<div class="cat-opt" data-name="${esc(c.name)}">${c.emoji||'📦'} ${esc(c.name)}</div>`

    opt.addEventListener('click', () => { $('fCat').value=opt.dataset.name; dd.classList.remove('open'); });

}
